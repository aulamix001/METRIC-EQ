cmake_minimum_required(VERSION 3.22)
project(CHiMetricMatchPro VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(CHiMetricMatchPro COMPANY_NAME "CHi Audio" BUNDLE_ID "audio.chi.metricmatchpro" PLUGIN_MANUFACTURER_CODE Chia PLUGIN_CODE Chmp FORMATS VST3 PRODUCT_NAME "CHi Metric Match Pro" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(CHiMetricMatchPro PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(CHiMetricMatchPro PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(CHiMetricMatchPro PRIVATE juce::juce_audio_processors juce::juce_dsp juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
include(CTest)
if(BUILD_TESTING)
 juce_add_console_app(CHiTests PRODUCT_NAME "CHi Foundation Tests")
 # The test executable links plugin shared code; use its standalone definition.
 get_target_property(test_definitions CHiTests COMPILE_DEFINITIONS)
 list(FILTER test_definitions EXCLUDE REGEX "^JUCE_STANDALONE_APPLICATION=")
 set_property(TARGET CHiTests PROPERTY COMPILE_DEFINITIONS "${test_definitions}")
 target_sources(CHiTests PRIVATE Tests/DSPTests.cpp)
 target_link_libraries(CHiTests PRIVATE CHiMetricMatchPro)
 target_include_directories(CHiTests PRIVATE ${juce_SOURCE_DIR}/modules)
 target_compile_definitions(CHiTests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
 add_test(NAME foundation COMMAND CHiTests)
 add_test(NAME parameter_mapping COMMAND CHiTests --mapping)
 add_test(NAME lifecycle COMMAND CHiTests --lifecycle)
 set_tests_properties(foundation parameter_mapping lifecycle PROPERTIES TIMEOUT 60)
endif()
